  <script>

    /* =====================================================
       ПЕРЕМЕННЫЕ ГАЛЕРЕИ
       ===================================================== */

    let currentImageIndex = 0;

    let currentImages = [];


    /* =====================================================
       DOM
       ===================================================== */

    const modal =
      document.getElementById(
        "productModal"
      );


    const accountModal =
      document.getElementById(
        "accountModal"
      );


    const galleryTrack =
      document.getElementById(
        "galleryTrack"
      );


    const galleryLeft =
      document.getElementById(
        "galleryLeft"
      );


    const galleryRight =
      document.getElementById(
        "galleryRight"
      );


    const galleryDots =
      document.getElementById(
        "galleryDots"
      );


    /* =====================================================
       АККАУНТ
       ===================================================== */

    function openAccountModal(event) {

      if (event) {

        event.stopPropagation();

      }


      accountModal.classList.add(
        "active"
      );


      document.body.style.overflow =
        "hidden";

    }


    /* =====================================================
       ЗАКРЫТИЕ АККАУНТА
       ===================================================== */

    function closeAccountModal(event) {


      /*
       * Если нажали внутри окна,
       * не закрываем.
       */

      if (
        event &&
        event.target !== event.currentTarget
      ) {

        return;

      }


      accountModal.classList.remove(
        "active"
      );


      document.body.style.overflow =
        "";

    }


    /* =====================================================
       СОЗДАНИЕ ГАЛЕРЕИ
       ===================================================== */

    function buildGallery() {


      galleryTrack.innerHTML = "";

      galleryDots.innerHTML = "";


      if (
        !currentImages ||
        currentImages.length === 0
      ) {

        return;

      }


      /* =================================================
         СОЗДАЁМ ФОТОГРАФИИ
         ================================================= */

      currentImages.forEach(
        function(imageUrl, index) {


          const slide =
            document.createElement(
              "div"
            );


          slide.className =
            "gallery-slide";


          const image =
            document.createElement(
              "img"
            );


          image.src =
            imageUrl;


          image.alt =
            document.getElementById(
              "modalTitle"
            ).textContent;


          image.draggable =
            false;


          slide.appendChild(
            image
          );


          galleryTrack.appendChild(
            slide
          );


          /* =============================================
             ТОЧКА
             ============================================= */

          const dot =
            document.createElement(
              "button"
            );


          dot.type =
            "button";


          dot.className =
            "gallery-dot";


          dot.setAttribute(
            "aria-label",
            "Фотография " + (index + 1)
          );


          dot.addEventListener(
            "click",
            function(event) {

              event.stopPropagation();

              showImage(index);

            }
          );


          galleryDots.appendChild(
            dot
          );

        }
      );


      if (
        currentImages.length <= 1
      ) {

        galleryDots.classList.add(
          "single"
        );

      } else {

        galleryDots.classList.remove(
          "single"
        );

      }


      updateDots();

    }


    /* =====================================================
       ОТКРЫТИЕ ТОВАРА
       ===================================================== */

    function openProduct(card) {


      const title =
        card.dataset.title ||
        "Без названия";


      const description =
        card.dataset.description ||
        "Описание отсутствует";


      const colors =
        card.dataset.colors ||
        "";


      const category =
        card.dataset.category ||
        "";


      const price =
        card.dataset.price ||
        "0";


      const kolvo =
        Number(
          card.dataset.kolvo || 0
        );


      /* =================================================
         ФОТОГРАФИИ
         ================================================= */

      let images = [];


      try {

        images =
          JSON.parse(
            card.dataset.images ||
            "[]"
          );

      } catch (error) {

        console.error(
          "Ошибка чтения фотографий:",
          error
        );

        images = [];

      }


      if (
        images.length === 0 &&
        card.dataset.image
      ) {

        images = [
          card.dataset.image
        ];

      }


      currentImages =
        images;


      currentImageIndex =
        0;


      /* =================================================
         ТЕКСТ
         ================================================= */

      document
        .getElementById(
          "modalTitle"
        )
        .textContent =
          title;


      document
        .getElementById(
          "modalDescription"
        )
        .textContent =
          description;


      /* =================================================
         КАТЕГОРИЯ
         ================================================= */

      const categoryElement =
        document.getElementById(
          "modalCategory"
        );


      categoryElement.textContent =
        category;


      if (category) {

        categoryElement.style.display =
          "block";

      } else {

        categoryElement.style.display =
          "none";

      }


      /* =================================================
         ЦВЕТ
         ================================================= */

      const colorsElement =
        document.getElementById(
          "modalColors"
        );


      if (colors) {

        colorsElement.textContent =
          "Цвет: " + colors;


        colorsElement.style.display =
          "block";

      } else {

        colorsElement.textContent =
          "";


        colorsElement.style.display =
          "none";

      }


      /* =================================================
         ЦЕНА
         ================================================= */

      document
        .getElementById(
          "modalPrice"
        )
        .textContent =
          price + " ₽";


      /* =================================================
         НАЛИЧИЕ
         ================================================= */

      const stockElement =
        document.getElementById(
          "modalStock"
        );


      if (kolvo > 0) {

        stockElement.textContent =
          "В наличии: " + kolvo;

      } else {

        stockElement.textContent =
          "Нет в наличии";

      }


      /* =================================================
         ГАЛЕРЕЯ
         ================================================= */

      buildGallery();


      /* =================================================
         ОТКРЫВАЕМ
         ================================================= */

      modal.classList.add(
        "active"
      );


      document.body.style.overflow =
        "hidden";


      requestAnimationFrame(
        function() {

          showImage(0);

        }
      );

    }


    /* =====================================================
       ПОКАЗ ФОТОГРАФИИ
       ===================================================== */

    function showImage(index) {


      if (
        !currentImages ||
        currentImages.length === 0
      ) {

        return;

      }


      /* =================================================
         ЗАЦИКЛИВАНИЕ
         ================================================= */

      if (
        index >= currentImages.length
      ) {

        index = 0;

      }


      if (index < 0) {

        index =
          currentImages.length - 1;

      }


      currentImageIndex =
        index;


      /* =================================================
         ПРОКРУТКА
         ================================================= */

      const slideWidth =
        galleryTrack.clientWidth;


      galleryTrack.scrollTo({

        left:
          slideWidth * currentImageIndex,

        behavior:
          "smooth"

      });


      updateDots();

    }


    /* =====================================================
       ОБНОВЛЕНИЕ ТОЧЕК
       ===================================================== */

    function updateDots() {


      const dots =
        galleryDots.querySelectorAll(
          ".gallery-dot"
        );


      dots.forEach(
        function(dot, index) {

          if (
            index === currentImageIndex
          ) {

            dot.classList.add(
              "active"
            );

          } else {

            dot.classList.remove(
              "active"
            );

          }

        }
      );

    }


    /* =====================================================
       ЛЕВАЯ ПОЛОВИНА
       ===================================================== */

    galleryLeft.addEventListener(
      "click",
      function(event) {

        event.stopPropagation();


        if (
          currentImages.length <= 1
        ) {

          return;

        }


        showImage(
          currentImageIndex - 1
        );

      }
    );


    /* =====================================================
       ПРАВАЯ ПОЛОВИНА
       ===================================================== */

    galleryRight.addEventListener(
      "click",
      function(event) {

        event.stopPropagation();


        if (
          currentImages.length <= 1
        ) {

          return;

        }


        showImage(
          currentImageIndex + 1
        );

      }
    );


    /* =====================================================
       ТОЧКИ
       ===================================================== */

    galleryDots.addEventListener(
      "click",
      function(event) {

        event.stopPropagation();

      }
    );


    /* =====================================================
       СВАЙП / РУЧНОЙ СКРОЛЛ
       ===================================================== */

    galleryTrack.addEventListener(
      "scroll",
      function() {

        if (
          !currentImages.length
        ) {

          return;

        }


        const slideWidth =
          galleryTrack.clientWidth;


        if (
          slideWidth <= 0
        ) {

          return;

        }


        const index =
          Math.round(
            galleryTrack.scrollLeft /
            slideWidth
          );


        if (
          index !== currentImageIndex
        ) {

          currentImageIndex =
            Math.max(
              0,
              Math.min(
                index,
                currentImages.length - 1
              )
            );


          updateDots();

        }

      },
      {
        passive: true
      }
    );


    /* =====================================================
       КЛАВИАТУРА ДЛЯ ТОВАРА
       ===================================================== */

    document.addEventListener(
      "keydown",
      function(event) {


        if (
          !modal.classList.contains(
            "active"
          )
        ) {

          return;

        }


        if (
          event.key === "ArrowRight"
        ) {

          event.preventDefault();


          showImage(
            currentImageIndex + 1
          );

        }


        if (
          event.key === "ArrowLeft"
        ) {

          event.preventDefault();


          showImage(
            currentImageIndex - 1
          );

        }

      }
    );


    /* =====================================================
       ЗАКРЫТИЕ ТОВАРА
       ===================================================== */

    function closeProduct(event) {

      // Если кликнули по фону модального окна —
      // закрываем.
      if (
        event &&
        event.target === event.currentTarget
      ) {
        modal.classList.remove("active");
        document.body.style.overflow = "";
        return;
      }

      // Если функция вызвана напрямую крестиком —
      // тоже закрываем.
      if (!event) {
        modal.classList.remove("active");
        document.body.style.overflow = "";
      }
    }


    /* =====================================================
       ESC
       ===================================================== */

    document.addEventListener(
      "keydown",
      function(event) {

        if (
          event.key !== "Escape"
        ) {

          return;

        }


        if (
          modal.classList.contains(
            "active"
          )
        ) {

          closeProduct();

        }


        if (
          accountModal.classList.contains(
            "active"
          )
        ) {

          closeAccountModal();

        }

      }
    );


    /* =====================================================
       RESIZE
       ===================================================== */

    window.addEventListener(
      "resize",
      function() {


        if (
          !modal.classList.contains(
            "active"
          )
        ) {

          return;

        }


        const slideWidth =
          galleryTrack.clientWidth;


        galleryTrack.scrollLeft =
          slideWidth *
          currentImageIndex;

      }
    );

      const logo = document.querySelector(".tennis-logo");

      function clamp(value, min, max) {
        return Math.max(min, Math.min(max, value));
      }

      function easeOutCubic(value) {
        return 1 - Math.pow(1 - value, 3);
      }

      function updateTarget() {

        /*
         * 0 — большой логотип по центру.
         * 1 — компактный логотип в верхней панели слева.
         *
         * Вся трансформация проходит постепенно за 260px
         * прокрутки, поэтому резкого скачка больше нет.
         */
        targetProgress =
          clamp(window.scrollY / 260, 0, 1);

        document.body.classList.toggle(
          "scrolled",
          targetProgress > 0.18
        );

        startAnimation();
      }

      function startAnimation() {
        if (animationFrame !== null) {
          return;
        }

        animationFrame =
          requestAnimationFrame(animate);
      }

      function animate() {

        /*
         * Сглаживание делает движение мягким даже при
         * резком вращении колеса или движении трекпада.
         */
        progress +=
          (targetProgress - progress) * 0.16;

        if (
          Math.abs(targetProgress - progress) < 0.001
        ) {
          progress = targetProgress;
        }

        const p = easeOutCubic(progress);
        const viewportWidth = window.innerWidth;

        /* =================================================
